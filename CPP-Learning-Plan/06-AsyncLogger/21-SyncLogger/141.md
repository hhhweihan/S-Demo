  ```cpp
  enum class LogLevel { DEBUG, INFO, WARN, ERROR, FATAL };

  class Logger {
  public:
      static Logger& instance();
      void set_level(LogLevel lv) { level_ = lv; }
      void log(LogLevel lv, const char* file, int line,
               const std::string& msg);
  private:
      LogLevel level_ = LogLevel::INFO;

  };
  ```
